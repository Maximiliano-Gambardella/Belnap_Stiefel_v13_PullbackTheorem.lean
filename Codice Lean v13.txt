import Mathlib

set_option linter.unnecessarySeqFocus false

open Matrix

variable {rd d : ℕ}

/- ================= §2 — Definizioni ================= -/

noncomputable def A (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin d) (Fin d) ℂ :=
  Vᴴ * M * V

noncomputable def g (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin d) (Fin d) ℂ :=
  A M V - 1

noncomputable def e1 (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin d) (Fin d) ℂ :=
  Dᴴ * M * V + Vᴴ * M * D

noncomputable def e2 (M : Matrix (Fin rd) (Fin rd) ℂ) (D : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin d) (Fin d) ℂ :=
  Dᴴ * M * D

-- f(V) := Re tr(g(V)^2) = ‖g(V)‖²_F
noncomputable def f (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) : ℝ :=
  (Matrix.trace (g M V * g M V)).re

/- ================= §3.1 — Hermiticità ================= -/

theorem A_isHermitian (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V : Matrix (Fin rd) (Fin d) ℂ) :
    (A M V)ᴴ = A M V := by
  show (Vᴴ * M * V)ᴴ = Vᴴ * M * V
  rw [conjTranspose_mul, conjTranspose_mul, conjTranspose_conjTranspose,
    ← Matrix.mul_assoc, hM]

theorem g_isHermitian (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V : Matrix (Fin rd) (Fin d) ℂ) :
    (g M V)ᴴ = g M V := by
  show (A M V - 1)ᴴ = A M V - 1
  rw [conjTranspose_sub, A_isHermitian M hM V, conjTranspose_one]

theorem e1_isHermitian (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    (e1 M V D)ᴴ = e1 M V D := by
  show (Dᴴ * M * V + Vᴴ * M * D)ᴴ = Dᴴ * M * V + Vᴴ * M * D
  rw [conjTranspose_add, conjTranspose_mul, conjTranspose_mul,
    conjTranspose_mul, conjTranspose_mul,
    conjTranspose_conjTranspose, conjTranspose_conjTranspose, hM]
  rw [add_comm]
  simp only [Matrix.mul_assoc]

theorem e2_isHermitian (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (D : Matrix (Fin rd) (Fin d) ℂ) :
    (e2 M D)ᴴ = e2 M D := by
  show (Dᴴ * M * D)ᴴ = Dᴴ * M * D
  rw [conjTranspose_mul, conjTranspose_mul, conjTranspose_conjTranspose, hM]
  simp only [Matrix.mul_assoc]

/- ================= §3.2 — Espansione algebrica ================= -/

theorem A_expand (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ)
    (t : ℂ) :
    A M (V + t • D)
      = A M V + t • (Vᴴ * M * D) + (star t) • (Dᴴ * M * V)
        + (star t * t) • (Dᴴ * M * D) := by
  unfold A
  rw [conjTranspose_add, conjTranspose_smul]
  simp only [Matrix.add_mul, Matrix.mul_add, Matrix.smul_mul, Matrix.mul_smul, smul_add]
  module

theorem A_expand_real (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ)
    (t : ℝ) :
    A M (V + (t : ℂ) • D)
      = A M V + (t : ℂ) • (Dᴴ * M * V + Vᴴ * M * D) + (t ^ 2 : ℝ) • (Dᴴ * M * D) := by
  have ht : star (t : ℂ) = (t : ℂ) := by
    apply Complex.ext <;> simp
  rw [A_expand M V D (t : ℂ), ht]
  module

theorem g_expand_e (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ)
    (t : ℝ) :
    g M (V + (t : ℂ) • D)
      = g M V + (t : ℂ) • e1 M V D + (t ^ 2 : ℝ) • e2 M D := by
  unfold g e1 e2
  rw [A_expand_real M V D t]
  module

/- ================= §3.3 — Simmetria di traccia (fattore 4) ================= -/

theorem trace_re_symm (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    (Matrix.trace (g M V * (Vᴴ * M * D))).re
      = (Matrix.trace (g M V * (Dᴴ * M * V))).re := by
  have hconj : (g M V * (Dᴴ * M * V))ᴴ = Vᴴ * M * D * g M V := by
    rw [conjTranspose_mul, conjTranspose_mul, conjTranspose_mul,
      conjTranspose_conjTranspose, hM, g_isHermitian M hM V, ← Matrix.mul_assoc]
  have htrace : Matrix.trace ((g M V * (Dᴴ * M * V))ᴴ)
      = star (Matrix.trace (g M V * (Dᴴ * M * V))) :=
    Matrix.trace_conjTranspose _
  rw [hconj, Matrix.trace_mul_comm] at htrace
  have hre : (Matrix.trace (g M V * (Vᴴ * M * D))).re
      = (star (Matrix.trace (g M V * (Dᴴ * M * V)))).re := by
    rw [← htrace]
  rw [hre]
  show ((starRingEnd ℂ) (Matrix.trace (g M V * (Dᴴ * M * V)))).re
      = (Matrix.trace (g M V * (Dᴴ * M * V))).re
  exact Complex.conj_re _

theorem re_trace_g_e1 (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    (Matrix.trace (g M V * e1 M V D)).re
      = 2 * (Matrix.trace (g M V * (Vᴴ * M * D))).re := by
  unfold e1
  rw [mul_add, Matrix.trace_add]
  have key := trace_re_symm M hM V D
  show (Matrix.trace (g M V * (Dᴴ * M * V))
      + Matrix.trace (g M V * (Vᴴ * M * D))).re
    = 2 * (Matrix.trace (g M V * (Vᴴ * M * D))).re
  rw [Complex.add_re, ← key]
  ring

theorem trace_order1 (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix.trace (g M V * e1 M V D + e1 M V D * g M V)
      = 2 * Matrix.trace (g M V * e1 M V D) := by
  rw [Matrix.trace_add, Matrix.trace_mul_comm (e1 M V D) (g M V)]
  ring

theorem f_linear_coeff (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    (Matrix.trace (g M V * e1 M V D + e1 M V D * g M V)).re
      = 4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re := by
  rw [trace_order1]
  have h2 : (2 * Matrix.trace (g M V * e1 M V D)).re
      = 2 * (Matrix.trace (g M V * e1 M V D)).re := by
    simp [Complex.mul_re]
  have hstep := re_trace_g_e1 M hM V D
  rw [h2, hstep]
  ring

/- ================= §3.4 — Espansione polinomiale e derivata ================= -/

theorem gsq_expand (M : Matrix (Fin rd) (Fin rd) ℂ) (V D : Matrix (Fin rd) (Fin d) ℂ)
    (t : ℝ) :
    g M (V + (t : ℂ) • D) * g M (V + (t : ℂ) • D)
      = g M V * g M V
        + (t : ℂ) • (g M V * e1 M V D + e1 M V D * g M V)
        + (t : ℂ) ^ 2 • (g M V * e2 M D + e2 M D * g M V + e1 M V D * e1 M V D)
        + (t : ℂ) ^ 3 • (e1 M V D * e2 M D + e2 M D * e1 M V D)
        + (t : ℂ) ^ 4 • (e2 M D * e2 M D) := by
  rw [g_expand_e M V D t]
  simp only [add_mul, mul_add, smul_mul_assoc, mul_smul_comm]
  module

private lemma cRe_real_smul (t : ℝ) (z : ℂ) : ((t : ℂ) • z).re = t * z.re := by
  rw [smul_eq_mul, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im]
  ring

private lemma cRe_smul_ofReal_pow (t : ℝ) (k : ℕ) (z : ℂ) :
    (((t : ℂ) ^ k) • z).re = t ^ k * z.re := by
  rw [smul_eq_mul]
  have hcast : (t : ℂ) ^ k = ((t ^ k : ℝ) : ℂ) := by push_cast; ring
  rw [hcast, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im]
  ring

theorem f_expand_poly (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) (t : ℝ) :
    f M (V + (t : ℂ) • D)
      = f M V
        + t * (4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re)
        + t ^ 2 * (Matrix.trace
            (g M V * e2 M D + e2 M D * g M V + e1 M V D * e1 M V D)).re
        + t ^ 3 * (Matrix.trace
            (e1 M V D * e2 M D + e2 M D * e1 M V D)).re
        + t ^ 4 * (Matrix.trace (e2 M D * e2 M D)).re := by
  unfold f
  rw [gsq_expand M V D t]
  rw [Matrix.trace_add, Matrix.trace_add, Matrix.trace_add, Matrix.trace_add]
  rw [Matrix.trace_smul, Matrix.trace_smul, Matrix.trace_smul, Matrix.trace_smul]
  rw [Complex.add_re, Complex.add_re, Complex.add_re, Complex.add_re]
  rw [cRe_real_smul, cRe_smul_ofReal_pow, cRe_smul_ofReal_pow, cRe_smul_ofReal_pow]
  rw [f_linear_coeff M hM V D]

theorem hasDerivAt_poly4 (c0 c1 c2 c3 c4 : ℝ) :
    HasDerivAt (fun t : ℝ => c0 + t * c1 + t ^ 2 * c2 + t ^ 3 * c3 + t ^ 4 * c4) c1 0 := by
  have h0 : HasDerivAt (fun _ : ℝ => c0) 0 0 := hasDerivAt_const 0 c0
  have h1 : HasDerivAt (fun t : ℝ => t * c1) c1 0 := by
    simpa using (hasDerivAt_id (0 : ℝ)).mul_const c1
  have h2 : HasDerivAt (fun t : ℝ => t ^ 2 * c2) 0 0 := by
    simpa using (hasDerivAt_pow 2 (0 : ℝ)).mul_const c2
  have h3 : HasDerivAt (fun t : ℝ => t ^ 3 * c3) 0 0 := by
    simpa using (hasDerivAt_pow 3 (0 : ℝ)).mul_const c3
  have h4 : HasDerivAt (fun t : ℝ => t ^ 4 * c4) 0 0 := by
    simpa using (hasDerivAt_pow 4 (0 : ℝ)).mul_const c4
  have h01 : HasDerivAt (fun t : ℝ => c0 + t * c1) (0 + c1) 0 := h0.add h1
  have h012 : HasDerivAt (fun t : ℝ => c0 + t * c1 + t ^ 2 * c2) (0 + c1 + 0) 0 :=
    h01.add h2
  have h0123 : HasDerivAt (fun t : ℝ => c0 + t * c1 + t ^ 2 * c2 + t ^ 3 * c3)
      (0 + c1 + 0 + 0) 0 := h012.add h3
  have h01234 : HasDerivAt (fun t : ℝ => c0 + t * c1 + t ^ 2 * c2 + t ^ 3 * c3 + t ^ 4 * c4)
      (0 + c1 + 0 + 0 + 0) 0 := h0123.add h4
  have hval : (0 : ℝ) + c1 + 0 + 0 + 0 = c1 := by ring
  rwa [hval] at h01234

theorem f_hasDerivAt (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    HasDerivAt (fun t : ℝ => f M (V + (t : ℂ) • D))
      (4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re) 0 := by
  have hpoly := hasDerivAt_poly4 (f M V)
    (4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re)
    ((Matrix.trace (g M V * e2 M D + e2 M D * g M V + e1 M V D * e1 M V D)).re)
    ((Matrix.trace (e1 M V D * e2 M D + e2 M D * e1 M V D)).re)
    ((Matrix.trace (e2 M D * e2 M D)).re)
  have heq : (fun t : ℝ => f M (V + (t : ℂ) • D))
      = (fun t : ℝ => f M V
          + t * (4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re)
          + t ^ 2 * (Matrix.trace
              (g M V * e2 M D + e2 M D * g M V + e1 M V D * e1 M V D)).re
          + t ^ 3 * (Matrix.trace
              (e1 M V D * e2 M D + e2 M D * e1 M V D)).re
          + t ^ 4 * (Matrix.trace (e2 M D * e2 M D)).re) := by
    funext t
    exact f_expand_poly M hM V D t
  rw [heq]
  exact hpoly

/- ================= §4 — Spazio tangente ================= -/

def IsTangent (V D : Matrix (Fin rd) (Fin d) ℂ) : Prop :=
  Vᴴ * D + Dᴴ * V = 0

noncomputable def hermPart (X : Matrix (Fin d) (Fin d) ℂ) : Matrix (Fin d) (Fin d) ℂ :=
  (1 / 2 : ℂ) • (X + Xᴴ)

noncomputable def projStiefel (V Z : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin rd) (Fin d) ℂ :=
  Z - V * hermPart (Vᴴ * Z)

theorem tangent_skew (V D : Matrix (Fin rd) (Fin d) ℂ) (h : IsTangent V D) :
    (Vᴴ * D)ᴴ = -(Vᴴ * D) := by
  unfold IsTangent at h
  have h' : Dᴴ * V + Vᴴ * D = 0 := by rw [add_comm]; exact h
  have heq : Dᴴ * V = -(Vᴴ * D) := eq_neg_of_add_eq_zero_left h'
  rw [conjTranspose_mul, conjTranspose_conjTranspose, heq]

theorem herm_zero_of_skew (X : Matrix (Fin d) (Fin d) ℂ) (h : Xᴴ = -X) :
    hermPart X = 0 := by
  unfold hermPart
  rw [h]
  module

theorem proj_tangent_eq (V D : Matrix (Fin rd) (Fin d) ℂ) (h : IsTangent V D) :
    projStiefel V D = D := by
  unfold projStiefel
  rw [herm_zero_of_skew _ (tangent_skew V D h)]
  simp

/- ================= §4.3 — Autoaggiuntezza della proiezione (NUOVO) ================= -/

theorem hermPart_isHermitian (X : Matrix (Fin d) (Fin d) ℂ) :
    (hermPart X)ᴴ = hermPart X := by
  unfold hermPart
  rw [conjTranspose_smul, conjTranspose_add, conjTranspose_conjTranspose]
  have hs : star (1 / 2 : ℂ) = (1 / 2 : ℂ) := by
    apply Complex.ext <;> simp
  rw [hs, add_comm]

theorem herm_trace_symm (X Y : Matrix (Fin d) (Fin d) ℂ) :
    (Matrix.trace (hermPart X * Y)).re = (Matrix.trace (Xᴴ * hermPart Y)).re := by
  have half_eq : (1 / 2 : ℂ) = ((1 / 2 : ℝ) : ℂ) := by
    apply Complex.ext <;> simp
  have e1 : Matrix.trace (hermPart X * Y)
      = (1 / 2 : ℂ) • (Matrix.trace (X * Y) + Matrix.trace (Xᴴ * Y)) := by
    unfold hermPart
    rw [smul_mul_assoc, add_mul, Matrix.trace_smul, Matrix.trace_add]
  have e2 : Matrix.trace (Xᴴ * hermPart Y)
      = (1 / 2 : ℂ) • (Matrix.trace (Xᴴ * Y) + Matrix.trace (Xᴴ * Yᴴ)) := by
    unfold hermPart
    rw [mul_smul_comm, mul_add, Matrix.trace_smul, Matrix.trace_add]
  have e3 : Matrix.trace (Xᴴ * Yᴴ) = star (Matrix.trace (X * Y)) := by
    have hXY : Xᴴ * Yᴴ = (Y * X)ᴴ := by rw [conjTranspose_mul]
    rw [hXY, Matrix.trace_conjTranspose, Matrix.trace_mul_comm]
  have hre : (Matrix.trace (Xᴴ * Yᴴ)).re = (Matrix.trace (X * Y)).re := by
    rw [e3]
    exact Complex.conj_re _
  rw [e1, e2, half_eq]
  rw [smul_eq_mul, smul_eq_mul, Complex.mul_re, Complex.mul_re, Complex.ofReal_re,
    Complex.ofReal_im, Complex.add_re, Complex.add_re, hre]
  ring

noncomputable def innerTr (X Y : Matrix (Fin rd) (Fin d) ℂ) : ℝ :=
  (Matrix.trace (Xᴴ * Y)).re

theorem projStiefel_selfAdjoint (V X Y : Matrix (Fin rd) (Fin d) ℂ) :
    innerTr (projStiefel V X) Y = innerTr X (projStiefel V Y) := by
  unfold innerTr projStiefel
  have hconj : (X - V * hermPart (Vᴴ * X))ᴴ = Xᴴ - hermPart (Vᴴ * X) * Vᴴ := by
    rw [conjTranspose_sub, conjTranspose_mul, hermPart_isHermitian]
  rw [hconj, Matrix.sub_mul, Matrix.mul_sub, Matrix.trace_sub, Matrix.trace_sub,
    Complex.sub_re, Complex.sub_re]
  have step1 : hermPart (Vᴴ * X) * Vᴴ * Y = hermPart (Vᴴ * X) * (Vᴴ * Y) := by
    rw [Matrix.mul_assoc]
  have step2 : Xᴴ * (V * hermPart (Vᴴ * Y)) = (Vᴴ * X)ᴴ * hermPart (Vᴴ * Y) := by
    rw [conjTranspose_mul, conjTranspose_conjTranspose, Matrix.mul_assoc]
  rw [step1, step2, herm_trace_symm (Vᴴ * X) (Vᴴ * Y)]

/- ================= Passo A — Tangenza della derivata sotto il vincolo di Stiefel ================= -/

-- Derivata di `star` su ℂ, come funzione di una variabile reale: da HasDerivAt h h' t
-- segue HasDerivAt (star ∘ h) (star h') t. Usa star z = 2 * z.re - z (identità elementare)
-- per ridursi alla derivata di `re`, evitando di trattare `im` separatamente.
private lemma hasDerivAt_star_complex {h : ℝ → ℂ} {h' : ℂ} {t : ℝ}
    (hh : HasDerivAt h h' t) :
    HasDerivAt (fun s => star (h s)) (star h') t := by
  have hre0 : HasDerivAt (fun s => (h s).re) h'.re t :=
    Complex.reCLM.hasFDerivAt.comp_hasDerivAt t hh
  have hre : HasDerivAt (fun s => ((h s).re : ℂ)) ((h'.re : ℂ)) t :=
    Complex.ofRealCLM.hasFDerivAt.comp_hasDerivAt t hre0
  have heq : (fun s => star (h s)) = fun s => (2 : ℂ) * ((h s).re : ℂ) - h s := by
    funext s
    apply Complex.ext <;> simp <;> ring
  have hval : star h' = (2 : ℂ) * (h'.re : ℂ) - h' := by
    apply Complex.ext <;> simp <;> ring
  rw [heq, hval]
  exact (hre.const_mul (2 : ℂ)).sub hh

-- Derivata della trasposta coniugata: da HasDerivAt γ D t segue HasDerivAt (γᴴ) Dᴴ t,
-- dimostrata riducendo entrata per entrata via hasDerivAt_pi. Nota: si usa
-- `refine hasDerivAt_pi.mpr fun i => ?_` (unificazione) e non `rw [hasDerivAt_pi]`
-- (pattern matching sintattico), perché le istanze di Matrix sono defeq ma non
-- sintatticamente identiche a quelle di Pi.
theorem HasDerivAt.matrix_conjTranspose {p q : ℕ} {γ : ℝ → Matrix (Fin p) (Fin q) ℂ}
    {D : Matrix (Fin p) (Fin q) ℂ} {t : ℝ} (hγ : HasDerivAt γ D t) :
    HasDerivAt (fun s => (γ s)ᴴ) Dᴴ t := by
  have hγ' : ∀ i k, HasDerivAt (fun s => γ s i k) (D i k) t := fun i =>
    hasDerivAt_pi.mp (hasDerivAt_pi.mp hγ i)
  refine hasDerivAt_pi.mpr fun j => hasDerivAt_pi.mpr fun i => ?_
  have hval : (fun s => (γ s)ᴴ j i) = fun s => star (γ s i j) := by
    funext s
    exact Matrix.conjTranspose_apply (γ s) i j
  have hval' : Dᴴ j i = star (D i j) := Matrix.conjTranspose_apply D i j
  rw [hval, hval']
  exact hasDerivAt_star_complex (hγ' i j)

-- Regola del prodotto per matrici (rettangolari, dimensioni compatibili), dimostrata
-- entrata per entrata via hasDerivAt_pi + Matrix.mul_apply + HasDerivAt.sum. Il termine
-- sommato va isolato in un `have` con tipo esplicito PRIMA di passarlo a HasDerivAt.sum,
-- altrimenti l'inferenza higher-order dell'implicito `f : ι → ℝ → F` va in confusione.
theorem HasDerivAt.matrix_mul {p q r : ℕ} {γ : ℝ → Matrix (Fin p) (Fin q) ℂ}
    {δ : ℝ → Matrix (Fin q) (Fin r) ℂ} {Dγ : Matrix (Fin p) (Fin q) ℂ}
    {Dδ : Matrix (Fin q) (Fin r) ℂ} {t : ℝ}
    (hγ : HasDerivAt γ Dγ t) (hδ : HasDerivAt δ Dδ t) :
    HasDerivAt (fun s => γ s * δ s) (Dγ * δ t + γ t * Dδ) t := by
  have hγ' : ∀ i k, HasDerivAt (fun s => γ s i k) (Dγ i k) t := fun i =>
    hasDerivAt_pi.mp (hasDerivAt_pi.mp hγ i)
  have hδ' : ∀ k j, HasDerivAt (fun s => δ s k j) (Dδ k j) t := fun k =>
    hasDerivAt_pi.mp (hasDerivAt_pi.mp hδ k)
  refine hasDerivAt_pi.mpr fun i => hasDerivAt_pi.mpr fun j => ?_
  have hterm : ∀ k : Fin q, HasDerivAt (fun s => γ s i k * δ s k j)
      (Dγ i k * δ t k j + γ t i k * Dδ k j) t := fun k => (hγ' i k).mul (hδ' k j)
  have hij0 : HasDerivAt (∑ k, fun s => γ s i k * δ s k j)
      (∑ k, (Dγ i k * δ t k j + γ t i k * Dδ k j)) t :=
    HasDerivAt.sum (fun k _ => hterm k)
  have heq2 : (∑ k : Fin q, fun s => γ s i k * δ s k j)
      = fun s => ∑ k, γ s i k * δ s k j := by
    funext s
    rw [Finset.sum_apply]
  rw [heq2] at hij0
  have hval : (fun s => (γ s * δ s) i j) = fun s => ∑ k, γ s i k * δ s k j := by
    funext s
    exact Matrix.mul_apply
  have hval' : (Dγ * δ t + γ t * Dδ) i j
      = ∑ k, (Dγ i k * δ t k j + γ t i k * Dδ k j) := by
    rw [Matrix.add_apply, Matrix.mul_apply, Matrix.mul_apply, Finset.sum_add_distrib]
  rw [hval, hval']
  exact hij0

-- Unicità della derivata per funzioni a valori Matrice: Matrix non ha un'istanza
-- NormedAddCommGroup globale (solo le istanze generiche di modulo/topologia usate
-- dalla definizione base di HasDerivAt), quindi `HasDerivAt.unique` (che richiede
-- Normed) non si applica direttamente a livello di matrice. Si riduce entrata per
-- entrata a ℂ, dove NormedAddCommGroup è sempre disponibile.
theorem HasDerivAt.matrix_unique {p q : ℕ} {γ : ℝ → Matrix (Fin p) (Fin q) ℂ}
    {D₁ D₂ : Matrix (Fin p) (Fin q) ℂ} {t : ℝ}
    (h1 : HasDerivAt γ D₁ t) (h2 : HasDerivAt γ D₂ t) : D₁ = D₂ := by
  have h1' : ∀ i k, HasDerivAt (fun s => γ s i k) (D₁ i k) t := fun i =>
    hasDerivAt_pi.mp (hasDerivAt_pi.mp h1 i)
  have h2' : ∀ i k, HasDerivAt (fun s => γ s i k) (D₂ i k) t := fun i =>
    hasDerivAt_pi.mp (hasDerivAt_pi.mp h2 i)
  apply Matrix.ext
  intro i k
  exact (h1' i k).unique (h2' i k)

-- Teorema centrale del Passo A: se γ(t) ∈ St(d,n) per ogni t (vincolo Aᴴ*A = I_d
-- identicamente lungo la curva), allora la derivata di γ in t₀ è tangente a St(d,n)
-- in γ(t₀). Applicato a γ(t) = Φ(θ + t·eᵢ), questo è precisamente
-- "∂θᵢΦ(θ) ∈ T_{Φ(θ)}St" quando Φ(θ) ∈ St(d,n) per ogni θ.
theorem stiefel_deriv_isTangent {Φ : ℝ → Matrix (Fin rd) (Fin d) ℂ}
    {D : Matrix (Fin rd) (Fin d) ℂ} {t₀ : ℝ}
    (hSt : ∀ t, (Φ t)ᴴ * Φ t = 1)
    (hΦ : HasDerivAt Φ D t₀) :
    IsTangent (Φ t₀) D := by
  have h1 : HasDerivAt (fun s => (Φ s)ᴴ * Φ s) (Dᴴ * Φ t₀ + (Φ t₀)ᴴ * D) t₀ :=
    hΦ.matrix_conjTranspose.matrix_mul hΦ
  have h2 : HasDerivAt (fun s => (Φ s)ᴴ * Φ s) 0 t₀ := by
    have hconst : (fun s => (Φ s)ᴴ * Φ s) = fun _ : ℝ => (1 : Matrix (Fin d) (Fin d) ℂ) :=
      funext hSt
    rw [hconst]
    refine hasDerivAt_pi.mpr fun i => hasDerivAt_pi.mpr fun j => ?_
    exact hasDerivAt_const t₀ ((1 : Matrix (Fin d) (Fin d) ℂ) i j)
  have hzero : Dᴴ * Φ t₀ + (Φ t₀)ᴴ * D = 0 := h1.matrix_unique h2
  unfold IsTangent
  rw [add_comm]
  exact hzero

/- ================= §4.5 — Forma chiusa del gradiente Riemanniano (NUOVO) =================
   Obiettivo: grad_St f(V) = 4 (I_n − V V†) M V (A(V) − I_d).
   Strategia:
   1. A(V) e g(V) commutano sempre (identità algebrica pura, g = A − 1).
   2. Quindi A(V)·g(V) è Hermitiana (essendo A e g entrambe Hermitiane e commutanti).
   3. Il gradiente Euclideo è ∇f(V) = 4·M·V·g(V) (dal teorema f_hasDerivAt, §3.4,
      via caratterizzazione di Riesz: dfV(Δ) = ⟨∇f(V), Δ⟩ = 4 ℜ tr(g(V)·V†MΔ)).
   4. Proiettando su T_V St: dato che V†(4MVg(V)) = 4 A(V) g(V) è già Hermitiana,
      Herm(V†∇f(V)) = V†∇f(V) stesso, e quindi
      grad_St f(V) = ∇f(V) − V·(V†∇f(V)) = 4MVg(V) − 4V·A(V)·g(V)
                   = 4 (MV − V·V†MV) g(V) = 4 (I_n − VV†) M V (A(V) − I_d). -/

-- A(V) e g(V) commutano sempre, essendo g(V) = A(V) − 1: identità puramente
-- algebrica in un anello (anche non commutativo), non richiede Hermiticità.
theorem A_comm_g (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    A M V * g M V = g M V * A M V := by
  unfold g
  noncomm_ring

-- Poiché A(V) e g(V) sono entrambe Hermitiane e commutano, il loro prodotto
-- A(V)·g(V) è Hermitiano.
theorem Ag_isHermitian (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V : Matrix (Fin rd) (Fin d) ℂ) :
    (A M V * g M V)ᴴ = A M V * g M V := by
  rw [conjTranspose_mul, g_isHermitian M hM V, A_isHermitian M hM V]
  exact (A_comm_g M V).symm

-- Per uno scalare complesso c reale (star c = c) e una matrice Hermitiana X,
-- la parte Hermitiana di c•X è semplicemente c•X.
theorem hermPart_smul_real (c : ℂ) (hc : star c = c) (X : Matrix (Fin d) (Fin d) ℂ)
    (hX : Xᴴ = X) : hermPart (c • X) = c • X := by
  unfold hermPart
  rw [conjTranspose_smul, hX, hc]
  module

-- Gradiente Euclideo di f in V: ∇f(V) = 4·M·V·g(V) (coefficiente dimostrato in
-- f_hasDerivAt, §3.4, come derivata direzionale genuina per ogni direzione Δ).
noncomputable def gradEuclid (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin rd) (Fin d) ℂ :=
  (4 : ℂ) • (M * V * g M V)

-- Gradiente Riemanniano: proiezione del gradiente Euclideo sullo spazio tangente
-- di St(d,n) in V, tramite l'operatore projStiefel definito in §4.
noncomputable def gradStiefel (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    Matrix (Fin rd) (Fin d) ℂ :=
  projStiefel V (gradEuclid M V)

-- Identità algebrica pura (associatività + distributività della moltiplicazione
-- di matrici), che collega M·V·g(V) − V·A(V)·g(V) alla forma fattorizzata
-- (I_n − V·V†)·M·V·g(V). Non richiede Hermiticità né il vincolo di Stiefel su V.
theorem stiefel_grad_algebra (M : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ) :
    M * V * g M V - V * (A M V * g M V) = (1 - V * Vᴴ) * M * V * g M V := by
  unfold A
  have h1 : V * (Vᴴ * M * V * g M V) = V * Vᴴ * M * V * g M V := by
    rw [← Matrix.mul_assoc V (Vᴴ * M * V) (g M V)]
    rw [← Matrix.mul_assoc V (Vᴴ * M) V]
    rw [← Matrix.mul_assoc V Vᴴ M]
  rw [h1, Matrix.sub_mul, Matrix.sub_mul, Matrix.sub_mul, Matrix.one_mul]

-- Teorema centrale: forma chiusa del gradiente Riemanniano su St(d,n),
-- grad_St f(V) = 4 (I_n − V V†) M V (A(V) − I_d).
theorem gradStiefel_eq (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V : Matrix (Fin rd) (Fin d) ℂ) :
    gradStiefel M V = (4 : ℂ) • ((1 - V * Vᴴ) * M * V * g M V) := by
  unfold gradStiefel projStiefel gradEuclid
  have hinner : Vᴴ * ((4 : ℂ) • (M * V * g M V)) = (4 : ℂ) • (A M V * g M V) := by
    rw [Matrix.mul_smul]
    congr 1
    unfold A
    rw [← Matrix.mul_assoc, ← Matrix.mul_assoc]
  rw [hinner]
  have hc4 : star (4 : ℂ) = (4 : ℂ) := by
    apply Complex.ext <;> simp
  rw [hermPart_smul_real (4 : ℂ) hc4 (A M V * g M V) (Ag_isHermitian M hM V)]
  rw [Matrix.mul_smul, ← smul_sub]
  congr 1
  exact stiefel_grad_algebra M V

/- ================= §4.6 — Collegamento con la Parameter-Shift Rule (NUOVO) =================
   Obiettivo: formalizzare il ponte algebrico tra la PSR applicata ad A(θ) (frequenza 1,
   quindi PSR a due punti esatta — fatto di analisi armonica sui gate, NON dimostrato qui,
   dipende dalla struttura dei generatori involutori: cfr. lavoro esterno
   QudeLeap/Lean-QuantumAlg, teorema trig_parameter_shift) e la derivata direzionale di
   f∘Φ (dimostrata in §3.4 come HasDerivAt genuino).

   Il risultato NON dimostra la PSR per A(θ) in sé (richiederebbe di modellare circuiti
   quantistici, stati, gate a due livelli — fuori dalla portata di questo file, che lavora
   solo con calcolo differenziale su matrici). Prende la PSR per A come ipotesi esplicita
   (hPSR) — esattamente il fatto che l'analisi preliminare del documento attribuisce al
   lavoro esterno — e ne deriva rigorosamente la formula chiusa per ∂θᵢ(f∘Φ):

     ∂θᵢ(f∘Φ)(θ) = tr(g(Φ(θ)) · (A(θ + (π/2)eᵢ) − A(θ − (π/2)eᵢ)))

   La derivazione usa:
   1. e1(M,V,D) è la derivata di A(V+tD) rispetto a t in t=0 (A_expand_real, §3.2):
      qui D gioca il ruolo di ∂θᵢΦ(θ), quindi e1(M,V,D) = ∂θᵢA(θ).
   2. La PSR standard a due punti include il fattore 1/2 caratteristico
      (∂E/∂θ = ½[E(θ+π/2) − E(θ−π/2)] per funzioni di frequenza 1): quindi
      hPSR assume e1(M,V,D) = ½•(A₊ − A₋).
   3. re_trace_g_e1 (§3.3) fornisce esattamente il fattore 2 complementare che,
      combinato con il ½ della PSR, elimina ogni coefficiente residuo. -/

-- Teorema del ponte PSR: la derivata direzionale di f, calcolata lungo la direzione D
-- (che rappresenta ∂θᵢΦ(θ)), è data esattamente dalla traccia di g(V) contro la
-- differenza PSR (A₊ − A₋), SENZA alcun coefficiente residuo — a patto che valga la
-- PSR per A lungo quella direzione (hPSR, fatto esterno non dimostrato in questo file).
theorem f_hasDerivAt_PSR (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) (Aplus Aminus : Matrix (Fin d) (Fin d) ℂ)
    (hPSR : e1 M V D = (1 / 2 : ℂ) • (Aplus - Aminus)) :
    HasDerivAt (fun t : ℝ => f M (V + (t : ℂ) • D))
      (Matrix.trace (g M V * (Aplus - Aminus))).re 0 := by
  have hderiv := f_hasDerivAt M hM V D
  have hcoeff : 4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re
      = (Matrix.trace (g M V * (Aplus - Aminus))).re := by
    have h1 := re_trace_g_e1 M hM V D
    have h2 : Matrix.trace (g M V * e1 M V D)
        = (1 / 2 : ℂ) • Matrix.trace (g M V * (Aplus - Aminus)) := by
      rw [hPSR, Matrix.mul_smul, Matrix.trace_smul]
    have half_eq : (1 / 2 : ℂ) = ((1 / 2 : ℝ) : ℂ) := by
      apply Complex.ext <;> simp
    have h2re : (Matrix.trace (g M V * e1 M V D)).re
        = (1 / 2 : ℝ) * (Matrix.trace (g M V * (Aplus - Aminus))).re := by
      rw [h2, half_eq, smul_eq_mul, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im]
      ring
    rw [h2re] at h1
    linarith [h1]
  rw [← hcoeff]
  exact hderiv

/- ================= §5 — Il Pullback Theorem (enunciato finale, NUOVO) =================
   Composizione dei risultati precedenti in un unico teorema:

     ⟨grad_St f(Φ(θ)), ∂θᵢΦ(θ)⟩ = tr(g(Φ(θ)) · (A(θ+(π/2)eᵢ) − A(θ−(π/2)eᵢ)))

   cioè: il prodotto scalare di traccia tra il gradiente Riemanniano (§4.5) e la
   direzione tangente D = ∂θᵢΦ(θ) (tangente per costruzione, §4.4, dato che Φ(θ) resta
   sulla varietà di Stiefel per ogni θ) coincide ESATTAMENTE con la quantità calcolabile
   su hardware quantistico via PSR a due punti su A(θ) (§4.6).

   Ingredienti:
   1. stiefel_deriv_isTangent (§4.4): D è tangente a St(d,n) in Φ(θ).
   2. proj_tangent_eq (§4) + projStiefel_selfAdjoint (§4.3): l'autoaggiuntezza della
      proiezione permette di spostare grad_St f = proj(∇f) su ∇f nel prodotto scalare,
      usando che proj(D) = D (D già tangente).
   3. gradEuclid_innerTr (nuovo, sotto): ⟨∇f(V), D⟩ = 4·tr(g(V)·(V†MD)) — il ponte
      algebrico tra il gradiente Euclideo come oggetto e il coefficiente di derivata
      già noto da f_hasDerivAt (§3.4).
   4. trace_PSR_eq (nuovo, sotto): la stessa identità algebrica già usata in
      f_hasDerivAt_PSR (§4.6), isolata qui come lemma riusabile senza toccare quel
      teorema già verificato. -/

-- Ponte algebrico: il prodotto scalare di traccia tra il gradiente Euclideo ∇f(V) e
-- una direzione D coincide con il coefficiente di derivata 4·tr(g(V)·(V†MD)) già
-- dimostrato in f_hasDerivAt (§3.4). Richiede solo Hermiticità di M e di g(V).
theorem gradEuclid_innerTr (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) :
    innerTr (gradEuclid M V) D = 4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re := by
  unfold innerTr gradEuclid
  have hconj : (M * V * g M V)ᴴ = g M V * Vᴴ * M := by
    rw [conjTranspose_mul, conjTranspose_mul, g_isHermitian M hM V, hM, ← Matrix.mul_assoc]
  have hc4 : star (4 : ℂ) = (4 : ℂ) := by
    apply Complex.ext <;> simp
  have hconj4 : ((4 : ℂ) • (M * V * g M V))ᴴ = (4 : ℂ) • (g M V * Vᴴ * M) := by
    rw [conjTranspose_smul, hconj, hc4]
  have hassoc : g M V * Vᴴ * M * D = g M V * (Vᴴ * M * D) := by
    rw [Matrix.mul_assoc (g M V) Vᴴ M, Matrix.mul_assoc (g M V) (Vᴴ * M) D]
  have h4eq : (4 : ℂ) = ((4 : ℝ) : ℂ) := by
    apply Complex.ext <;> simp
  rw [hconj4, Matrix.smul_mul, hassoc, Matrix.trace_smul, h4eq, smul_eq_mul,
      Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im]
  ring

-- Identità algebrica della PSR (stessa dimostrazione usata dentro f_hasDerivAt_PSR,
-- §4.6, isolata qui come lemma riusabile — non modifica quel teorema già verificato).
theorem trace_PSR_eq (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (V D : Matrix (Fin rd) (Fin d) ℂ) (Aplus Aminus : Matrix (Fin d) (Fin d) ℂ)
    (hPSR : e1 M V D = (1 / 2 : ℂ) • (Aplus - Aminus)) :
    (Matrix.trace (g M V * (Aplus - Aminus))).re
      = 4 * (Matrix.trace (g M V * (Vᴴ * M * D))).re := by
  have h1 := re_trace_g_e1 M hM V D
  have h2 : Matrix.trace (g M V * e1 M V D)
      = (1 / 2 : ℂ) • Matrix.trace (g M V * (Aplus - Aminus)) := by
    rw [hPSR, Matrix.mul_smul, Matrix.trace_smul]
  have half_eq : (1 / 2 : ℂ) = ((1 / 2 : ℝ) : ℂ) := by
    apply Complex.ext <;> simp
  have h2re : (Matrix.trace (g M V * e1 M V D)).re
      = (1 / 2 : ℝ) * (Matrix.trace (g M V * (Aplus - Aminus))).re := by
    rw [h2, half_eq, smul_eq_mul, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im]
    ring
  rw [h2re] at h1
  linarith [h1]

-- IL PULLBACK THEOREM (enunciato finale).
-- Sotto le ipotesi:
--  - Φ(t) ∈ St(d,n) per ogni t (hSt), cioè Φ resta sulla varietà lungo tutta la curva;
--  - D è la derivata di Φ in t₀ (hΦ), cioè D = ∂θᵢΦ(θ) nella notazione del documento;
--  - la PSR vale per A lungo questa direzione (hPSR — fatto esterno, cfr. §4.6);
-- allora il prodotto scalare di traccia tra il gradiente Riemanniano e D coincide
-- esattamente con la quantità calcolabile su hardware via PSR a due punti su A.
theorem pullback_theorem (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (Φ : ℝ → Matrix (Fin rd) (Fin d) ℂ) (D : Matrix (Fin rd) (Fin d) ℂ) (t₀ : ℝ)
    (hSt : ∀ t, (Φ t)ᴴ * Φ t = 1)
    (hΦ : HasDerivAt Φ D t₀)
    (Aplus Aminus : Matrix (Fin d) (Fin d) ℂ)
    (hPSR : e1 M (Φ t₀) D = (1 / 2 : ℂ) • (Aplus - Aminus)) :
    innerTr (gradStiefel M (Φ t₀)) D
      = (Matrix.trace (g M (Φ t₀) * (Aplus - Aminus))).re := by
  have hTangent : IsTangent (Φ t₀) D := stiefel_deriv_isTangent hSt hΦ
  have hProjD : projStiefel (Φ t₀) D = D := proj_tangent_eq (Φ t₀) D hTangent
  have hSelfAdj := projStiefel_selfAdjoint (Φ t₀) (gradEuclid M (Φ t₀)) D
  rw [hProjD] at hSelfAdj
  unfold gradStiefel
  rw [hSelfAdj, gradEuclid_innerTr M hM (Φ t₀) D]
  exact (trace_PSR_eq M hM (Φ t₀) D Aplus Aminus hPSR).symm

/- ================= §6 — Costruzione esplicita di Φ gate per gate (NUOVO) =================
   Obiettivo: trasformare le ipotesi astratte hSt, hΦ del pullback_theorem in teoremi
   derivati da una costruzione concreta a livello di gate, secondo lo schema
   Φ(θ) = U_L(θ_L) ⋯ U_1(θ_1) · V₀ con V₀ᴴV₀ = I_d e ogni U_i unitaria.

   Qui formalizziamo lo scheletro generale (chiusura di St(d,n) sotto l'azione
   unitaria, applicabile a QUALSIASI architettura — Clements, RBS, FBS — fornendo le
   rispettive matrici concrete) più un'istanza concreta minimale (rbsGate), che mostra
   che lo scheletro è non vacuo e correttamente strumentato. -/

-- Passo 3 dello schema: U unitaria preserva l'appartenenza a St(d,n).
-- (UV)†(UV) = V†(U†U)V = V†V = I_d.
theorem unitary_preserves_stiefel (U : Matrix (Fin rd) (Fin rd) ℂ) (V : Matrix (Fin rd) (Fin d) ℂ)
    (hU : Uᴴ * U = 1) (hV : Vᴴ * V = 1) :
    (U * V)ᴴ * (U * V) = 1 := by
  rw [conjTranspose_mul, Matrix.mul_assoc Vᴴ Uᴴ (U * V), ← Matrix.mul_assoc Uᴴ U V, hU,
      Matrix.one_mul, hV]

-- Passo induttivo dello schema, applicato a Φ(t) = U_after · γ(t) · U_before · V₀:
-- se U_before, U_after sono unitarie fisse, V₀ è un punto di Stiefel, e γ(t) è
-- unitaria per ogni t (il layer variabile, es. ∂θᵢ del gate i-esimo), allora Φ(t)
-- resta su St(d,n) per ogni t. Generalizza immediatamente a un numero qualsiasi di
-- layer fissi applicando ripetutamente unitary_preserves_stiefel.
theorem Phi_layered_isSt (Ubefore Uafter : Matrix (Fin rd) (Fin rd) ℂ)
    (hUbefore : Ubeforeᴴ * Ubefore = 1) (hUafter : Uafterᴴ * Uafter = 1)
    (V0 : Matrix (Fin rd) (Fin d) ℂ) (hV0 : V0ᴴ * V0 = 1)
    (γ : ℝ → Matrix (Fin rd) (Fin rd) ℂ) (hγ : ∀ t, (γ t)ᴴ * γ t = 1) :
    ∀ t, (Uafter * γ t * Ubefore * V0)ᴴ * (Uafter * γ t * Ubefore * V0) = 1 := by
  intro t
  have h1 : (Ubefore * V0)ᴴ * (Ubefore * V0) = 1 :=
    unitary_preserves_stiefel Ubefore V0 hUbefore hV0
  have h2 : (γ t * (Ubefore * V0))ᴴ * (γ t * (Ubefore * V0)) = 1 :=
    unitary_preserves_stiefel (γ t) (Ubefore * V0) (hγ t) h1
  have h3 : (Uafter * (γ t * (Ubefore * V0)))ᴴ * (Uafter * (γ t * (Ubefore * V0))) = 1 :=
    unitary_preserves_stiefel Uafter (γ t * (Ubefore * V0)) hUafter h2
  simpa [Matrix.mul_assoc] using h3

-- Costante-matrice ha derivata nulla, costruita entrata per entrata via hasDerivAt_pi
-- (esattamente come nel passo h2 di stiefel_deriv_isTangent, §4.4): necessario perché
-- Matrix (Fin p) (Fin q) ℂ rettangolare non ha un'istanza NormedAddCommGroup globale,
-- quindi hasDerivAt_const applicato all'intera matrice fallisce a sintetizzare l'istanza.
theorem hasDerivAt_matrix_const {p q : ℕ} (C : Matrix (Fin p) (Fin q) ℂ) (t : ℝ) :
    HasDerivAt (fun _ : ℝ => C) 0 t := by
  refine hasDerivAt_pi.mpr fun i => hasDerivAt_pi.mpr fun j => ?_
  exact hasDerivAt_const t (C i j)

-- Differenziabilità di Φ(t) = U_after · γ(t) · U_before · V₀: i layer fissi hanno
-- derivata nulla (costanti), il layer variabile γ contribuisce la sua derivata Dγ
-- (es. ∂θᵢU_i(θᵢ)), e HasDerivAt.matrix_mul (§4.4) propaga il tutto via regola del
-- prodotto, con i termini di derivata nulla che si annullano algebricamente.
theorem Phi_layered_hasDerivAt (Ubefore Uafter : Matrix (Fin rd) (Fin rd) ℂ)
    (V0 : Matrix (Fin rd) (Fin d) ℂ)
    (γ : ℝ → Matrix (Fin rd) (Fin rd) ℂ) (Dγ : Matrix (Fin rd) (Fin rd) ℂ) (t₀ : ℝ)
    (hγ : HasDerivAt γ Dγ t₀) :
    HasDerivAt (fun t => Uafter * γ t * Ubefore * V0) (Uafter * Dγ * Ubefore * V0) t₀ := by
  have hconst1 : HasDerivAt (fun _ : ℝ => Ubefore * V0) (0 : Matrix (Fin rd) (Fin d) ℂ) t₀ :=
    hasDerivAt_matrix_const (Ubefore * V0) t₀
  have hstep1 : HasDerivAt (fun t => γ t * (Ubefore * V0))
      (Dγ * (Ubefore * V0) + γ t₀ * (0 : Matrix (Fin rd) (Fin d) ℂ)) t₀ :=
    hγ.matrix_mul hconst1
  have hconstA : HasDerivAt (fun _ : ℝ => Uafter) (0 : Matrix (Fin rd) (Fin rd) ℂ) t₀ :=
    hasDerivAt_matrix_const Uafter t₀
  have hstep2 : HasDerivAt (fun t => Uafter * (γ t * (Ubefore * V0)))
      ((0 : Matrix (Fin rd) (Fin rd) ℂ) * (γ t₀ * (Ubefore * V0))
        + Uafter * (Dγ * (Ubefore * V0) + γ t₀ * (0 : Matrix (Fin rd) (Fin d) ℂ))) t₀ :=
    hconstA.matrix_mul hstep1
  have heq : (fun t => Uafter * (γ t * (Ubefore * V0))) = fun t => Uafter * γ t * Ubefore * V0 := by
    funext t
    rw [Matrix.mul_assoc, Matrix.mul_assoc]
  have hval : (0 : Matrix (Fin rd) (Fin rd) ℂ) * (γ t₀ * (Ubefore * V0))
      + Uafter * (Dγ * (Ubefore * V0) + γ t₀ * (0 : Matrix (Fin rd) (Fin d) ℂ))
      = Uafter * Dγ * Ubefore * V0 := by
    simp [Matrix.mul_assoc, Matrix.zero_mul, Matrix.mul_zero]
  rw [heq, hval] at hstep2
  exact hstep2

-- Istanza concreta minimale: un gate RBS (Reconfigurable Beam Splitter) a due livelli,
-- rappresentato come rotazione reale embedded in ℂ. Mostra che lo scheletro sopra
-- non è vacuo — la stessa tecnica si applica a qualunque architettura concreta
-- (Clements, FBS, ecc.) fornendo le rispettive matrici parametrizzate.
noncomputable def rbsGate (φ : ℝ) : Matrix (Fin 2) (Fin 2) ℂ :=
  !![(Real.cos φ : ℂ), -(Real.sin φ : ℂ); (Real.sin φ : ℂ), (Real.cos φ : ℂ)]

-- Passo 1 dello schema (unitarietà del gate): rbsGate(φ) è unitaria per ogni φ,
-- conseguenza diretta dell'identità trigonometrica fondamentale sin²+cos²=1.
theorem rbsGate_isUnitary (φ : ℝ) : (rbsGate φ)ᴴ * (rbsGate φ) = 1 := by
  have hscC : Complex.sin (φ : ℂ) ^ 2 + Complex.cos (φ : ℂ) ^ 2 = 1 := by
    have h := Real.sin_sq_add_cos_sq φ
    have e1 : Complex.sin (φ : ℂ) = (Real.sin φ : ℂ) := (Complex.ofReal_sin φ).symm
    have e2 : Complex.cos (φ : ℂ) = (Real.cos φ : ℂ) := (Complex.ofReal_cos φ).symm
    rw [e1, e2]
    exact_mod_cast h
  have hstar_cos : (starRingEnd ℂ) (Complex.cos (φ : ℂ)) = Complex.cos (φ : ℂ) := by
    rw [show Complex.cos (φ : ℂ) = (Real.cos φ : ℂ) from (Complex.ofReal_cos φ).symm]
    exact Complex.conj_ofReal _
  have hstar_sin : (starRingEnd ℂ) (Complex.sin (φ : ℂ)) = Complex.sin (φ : ℂ) := by
    rw [show Complex.sin (φ : ℂ) = (Real.sin φ : ℂ) from (Complex.ofReal_sin φ).symm]
    exact Complex.conj_ofReal _
  ext i j
  fin_cases i <;> fin_cases j <;>
    simp [rbsGate, Matrix.mul_apply, Fin.sum_univ_two, conjTranspose_apply,
          hstar_cos, hstar_sin] <;>
    first
      | linear_combination hscC
      | linear_combination (0 : ℂ) * hscC

-- Passo 4 dello schema (derivabilità del gate): le entrate di rbsGate(φ) sono
-- funzioni trigonometriche di φ, quindi HasDerivAt segue direttamente dalle derivate
-- standard di seno e coseno, componendo con l'inclusione ℝ ↪ ℂ (stesso schema di
-- hasDerivAt_star_complex, §4.4) ed entrata per entrata via hasDerivAt_pi.
theorem rbsGate_hasDerivAt (φ : ℝ) :
    HasDerivAt (fun t : ℝ => rbsGate (φ + t))
      (!![(-(Real.sin φ) : ℂ), -(Real.cos φ : ℂ); (Real.cos φ : ℂ), -(Real.sin φ : ℂ)]) 0 := by
  have hshift : HasDerivAt (fun t : ℝ => φ + t) 1 0 := by
    simpa using (hasDerivAt_id (0 : ℝ)).const_add φ
  have hcos : HasDerivAt (fun t : ℝ => Real.cos (φ + t)) (-(Real.sin φ)) 0 := by
    have h : HasDerivAt (fun t : ℝ => Real.cos (φ + t)) (-Real.sin (φ + 0) * 1) 0 := hshift.cos
    simpa using h
  have hsin : HasDerivAt (fun t : ℝ => Real.sin (φ + t)) (Real.cos φ) 0 := by
    have h : HasDerivAt (fun t : ℝ => Real.sin (φ + t)) (Real.cos (φ + 0) * 1) 0 := hshift.sin
    simpa using h
  have hcosC : HasDerivAt (fun t : ℝ => (Real.cos (φ + t) : ℂ)) ((-(Real.sin φ) : ℝ) : ℂ) 0 :=
    Complex.ofRealCLM.hasFDerivAt.comp_hasDerivAt 0 hcos
  have hsinC : HasDerivAt (fun t : ℝ => (Real.sin (φ + t) : ℂ)) ((Real.cos φ : ℝ) : ℂ) 0 :=
    Complex.ofRealCLM.hasFDerivAt.comp_hasDerivAt 0 hsin
  have hcosC' : HasDerivAt (fun t : ℝ => Complex.cos ((φ : ℂ) + (t : ℂ)))
      (-Complex.sin (φ : ℂ)) 0 := by
    have hfe : (fun t : ℝ => Complex.cos ((φ : ℂ) + (t : ℂ)))
        = fun t : ℝ => (Real.cos (φ + t) : ℂ) := by
      funext t
      rw [← Complex.ofReal_add, Complex.ofReal_cos]
    have hde : (-Complex.sin (φ : ℂ)) = ((-(Real.sin φ) : ℝ) : ℂ) := by
      rw [Complex.ofReal_neg, Complex.ofReal_sin]
    rw [hfe, hde]
    exact hcosC
  have hsinC' : HasDerivAt (fun t : ℝ => Complex.sin ((φ : ℂ) + (t : ℂ)))
      (Complex.cos (φ : ℂ)) 0 := by
    have hfe : (fun t : ℝ => Complex.sin ((φ : ℂ) + (t : ℂ)))
        = fun t : ℝ => (Real.sin (φ + t) : ℂ) := by
      funext t
      rw [← Complex.ofReal_add, Complex.ofReal_sin]
    have hde : (Complex.cos (φ : ℂ)) = ((Real.cos φ : ℝ) : ℂ) := by
      rw [Complex.ofReal_cos]
    rw [hfe, hde]
    exact hsinC
  refine hasDerivAt_pi.mpr fun i => hasDerivAt_pi.mpr fun j => ?_
  fin_cases i <;> fin_cases j <;>
    simp [rbsGate] <;>
    first
      | exact hcosC' | exact hsinC' | exact hcosC'.neg | exact hsinC'.neg

-- COROLLARIO FINALE: il Pullback Theorem interamente istanziato a partire da
-- ipotesi a livello di gate (nessuna ipotesi astratta hSt/hΦ residua).
theorem pullback_theorem_layered (M : Matrix (Fin rd) (Fin rd) ℂ) (hM : Mᴴ = M)
    (Ubefore Uafter : Matrix (Fin rd) (Fin rd) ℂ)
    (hUbefore : Ubeforeᴴ * Ubefore = 1) (hUafter : Uafterᴴ * Uafter = 1)
    (V0 : Matrix (Fin rd) (Fin d) ℂ) (hV0 : V0ᴴ * V0 = 1)
    (γ : ℝ → Matrix (Fin rd) (Fin rd) ℂ) (hγSt : ∀ t, (γ t)ᴴ * γ t = 1)
    (Dγ : Matrix (Fin rd) (Fin rd) ℂ) (t₀ : ℝ) (hγDeriv : HasDerivAt γ Dγ t₀)
    (Aplus Aminus : Matrix (Fin d) (Fin d) ℂ)
    (hPSR : e1 M (Uafter * γ t₀ * Ubefore * V0) (Uafter * Dγ * Ubefore * V0)
      = (1 / 2 : ℂ) • (Aplus - Aminus)) :
    innerTr (gradStiefel M (Uafter * γ t₀ * Ubefore * V0)) (Uafter * Dγ * Ubefore * V0)
      = (Matrix.trace (g M (Uafter * γ t₀ * Ubefore * V0) * (Aplus - Aminus))).re :=
  pullback_theorem M hM (fun t => Uafter * γ t * Ubefore * V0) (Uafter * Dγ * Ubefore * V0) t₀
    (Phi_layered_isSt Ubefore Uafter hUbefore hUafter V0 hV0 γ hγSt)
    (Phi_layered_hasDerivAt Ubefore Uafter V0 γ Dγ t₀ hγDeriv)
    Aplus Aminus hPSR

/- ================= §7 — Istanza concreta finale con rbsGate (NUOVO) =================
   Salda rbsGate (§6) con pullback_theorem_layered: nessuna ipotesi residua sul gate
   stesso resta libera — l'unica ipotesi non dimostrata in questo file è hPSR, il fatto
   di analisi armonica sui circuiti quantistici (frequenza 1 di A(θ)) che dipende dalla
   struttura fisica dei generatori involutori e non da algebra lineare pura. -/

-- Il Pullback Theorem, interamente concreto: Φ(θ) = U_after · rbsGate(θ) · U_before · V₀,
-- con U_before, U_after layer unitari fissi arbitrari e V₀ punto di Stiefel arbitrario.
-- L'unica ipotesi non discharged è hPSR (fatto esterno, cfr. §4.6).
theorem pullback_theorem_rbs (M : Matrix (Fin 2) (Fin 2) ℂ) (hM : Mᴴ = M)
    (Ubefore Uafter : Matrix (Fin 2) (Fin 2) ℂ)
    (hUbefore : Ubeforeᴴ * Ubefore = 1) (hUafter : Uafterᴴ * Uafter = 1)
    (V0 : Matrix (Fin 2) (Fin d) ℂ) (hV0 : V0ᴴ * V0 = 1)
    (θ : ℝ)
    (Aplus Aminus : Matrix (Fin d) (Fin d) ℂ)
    (hPSR : e1 M (Uafter * rbsGate θ * Ubefore * V0)
        (Uafter * !![(-(Real.sin θ) : ℂ), -(Real.cos θ : ℂ); (Real.cos θ : ℂ), -(Real.sin θ : ℂ)]
          * Ubefore * V0)
      = (1 / 2 : ℂ) • (Aplus - Aminus)) :
    innerTr (gradStiefel M (Uafter * rbsGate θ * Ubefore * V0))
        (Uafter * !![(-(Real.sin θ) : ℂ), -(Real.cos θ : ℂ); (Real.cos θ : ℂ), -(Real.sin θ : ℂ)]
          * Ubefore * V0)
      = (Matrix.trace (g M (Uafter * rbsGate θ * Ubefore * V0) * (Aplus - Aminus))).re := by
  have hγSt : ∀ t : ℝ, (rbsGate (θ + t))ᴴ * rbsGate (θ + t) = 1 := fun t => rbsGate_isUnitary (θ + t)
  have hγDeriv : HasDerivAt (fun t : ℝ => rbsGate (θ + t))
      (!![(-(Real.sin θ) : ℂ), -(Real.cos θ : ℂ); (Real.cos θ : ℂ), -(Real.sin θ : ℂ)]) 0 :=
    rbsGate_hasDerivAt θ
  have hPSR0 : e1 M (Uafter * rbsGate (θ + 0) * Ubefore * V0)
      (Uafter * !![(-(Real.sin θ) : ℂ), -(Real.cos θ : ℂ); (Real.cos θ : ℂ), -(Real.sin θ : ℂ)]
        * Ubefore * V0)
      = (1 / 2 : ℂ) • (Aplus - Aminus) := by
    simpa using hPSR
  have hmain := pullback_theorem_layered M hM Ubefore Uafter hUbefore hUafter V0 hV0
    (fun t => rbsGate (θ + t)) hγSt
    (!![(-(Real.sin θ) : ℂ), -(Real.cos θ : ℂ); (Real.cos θ : ℂ), -(Real.sin θ : ℂ)]) 0 hγDeriv
    Aplus Aminus hPSR0
  simpa using hmain
